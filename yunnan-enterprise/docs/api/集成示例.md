#### 2. 创建 API 模块

**文件：** `src/api/notice.js`

```javascript
import axios from 'axios';

// 配置基础 URL
const API_BASE_URL = import.meta.env.VITE_API_BASE_URL || 'http://localhost:8082';

// 创建 axios 实例
const apiClient = axios.create({
  baseURL: API_BASE_URL,
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json',
  },
});

/**
 * 获取通知列表
 * @param {Object} params - 查询参数
 * @param {string} params.enterprise_id - 企业ID
 * @param {number} params.page_no - 页码
 * @param {number} params.page_size - 每页数量
 * @returns {Promise<Object>} 通知列表
 */
export const fetchNoticeList = async (params) => {
  try {
    const response = await apiClient.get('/api/notice/list', { params });
    return response.data.data;
  } catch (error) {
    throw new Error(error.response?.data?.message || '获取通知列表失败');
  }
};

/**
 * 获取通知详情
 * @param {number} noticeId - 通知ID
 * @param {Object} params - 查询参数
 * @returns {Promise<Object>} 通知详情
 */
export const fetchNoticeDetail = async (noticeId, params) => {
  try {
    const response = await apiClient.get(`/api/notice/${noticeId}`, { params });
    return response.data.data;
  } catch (error) {
    throw new Error(error.response?.data?.message || '获取通知详情失败');
  }
};

/**
 * 获取未读数量
 * @param {string} username - 用户名
 * @returns {Promise<number>} 未读数量
 */
export const fetchUnreadCount = async (username) => {
  try {
    const response = await apiClient.get('/api/notice/unread/count', {
      params: { username },
    });
    return response.data.data;
  } catch (error) {
    throw new Error(error.response?.data?.message || '获取未读数量失败');
  }
};
```

### 通知列表组件

```vue
<template>
  <div class="notice-list">
    <!-- 页面头部 -->
    <div class="page-header">
      <h1>通知中心</h1>
      <el-badge :value="unreadCount" :max="99" v-if="unreadCount > 0">
        <el-icon><Bell /></el-icon>
      </el-badge>
    </div>

    <!-- 通知列表 -->
    <el-card v-loading="loading">
      <!-- 空状态 -->
      <el-empty v-if="!loading && noticeList.length === 0" description="暂无通知" />

      <!-- 列表项 -->
      <div class="notice-items" v-else>
        <div
          v-for="notice in noticeList"
          :key="notice.noticeId"
          class="notice-item"
          :class="{ important: notice.isImportant === 1, unread: !notice.isRead }"
          @click="handleViewNotice(notice)"
        >
          <!-- 重要标签 -->
          <el-tag v-if="notice.isImportant === 1" type="danger" size="small" effect="dark">
            重要
          </el-tag>

          <!-- 标题 -->
          <h3 class="notice-title">{{ notice.title }}</h3>

          <!-- 元信息 -->
          <div class="notice-meta">
            <span>{{ notice.publisher }}</span>
            <span>{{ formatDate(notice.publishTime) }}</span>
            <span v-if="notice.attachment">
              <el-icon><Paperclip /></el-icon> 附件
            </span>
          </div>
        </div>
      </div>

      <!-- 分页 -->
      <el-pagination
        v-if="total > pageSize"
        v-model:current-page="currentPage"
        :page-size="pageSize"
        :total="total"
        layout="prev, pager, next, total"
        @current-change="handlePageChange"
        class="pagination"
      />
    </el-card>
  </div>
</template>

<script setup>
import { onMounted, ref, onUnmounted } from 'vue';
import { useRouter } from 'vue-router';
import { ElMessage } from 'element-plus';
import { Bell, Paperclip } from '@element-plus/icons-vue';
import { fetchNoticeList, fetchUnreadCount } from '@/api/notice';

const router = useRouter();

// 响应式数据
const loading = ref(false);
const noticeList = ref([]);
const currentPage = ref(1);
const pageSize = ref(10);
const total = ref(0);
const unreadCount = ref(0);

// TODO: 从登录信息获取
const enterpriseId = ref('ENTERPRISE_001');
const username = ref('test_user');

// 定时器
let refreshTimer = null;

/**
 * 加载通知列表
 */
const loadNoticeList = async () => {
  loading.value = true;
  try {
    const data = await fetchNoticeList({
      enterprise_id: enterpriseId.value,
      page_no: currentPage.value,
      page_size: pageSize.value,
    });
    noticeList.value = data.list || [];
    total.value = data.totalCount || 0;
  } catch (error) {
    ElMessage.error(error?.message || '加载通知列表失败');
  } finally {
    loading.value = false;
  }
};

/**
 * 加载未读数量
 */
const loadUnreadCount = async () => {
  try {
    const count = await fetchUnreadCount(username.value);
    unreadCount.value = count;
  } catch (error) {
    console.error('获取未读数量失败', error);
  }
};

/**
 * 查看通知详情
 */
const handleViewNotice = (notice) => {
  router.push({
    path: '/notice/detail',
    query: { id: notice.noticeId },
  });
};

/**
 * 页码改变
 */
const handlePageChange = (page) => {
  currentPage.value = page;
  loadNoticeList();
};

/**
 * 格式化日期
 */
const formatDate = (dateStr) => {
  if (!dateStr) return '-';
  const date = new Date(dateStr);
  const now = new Date();
  const diff = now - date;
  const minutes = Math.floor(diff / 60000);
  const hours = Math.floor(diff / 3600000);
  const days = Math.floor(diff / 86400000);

  if (minutes < 1) return '刚刚';
  if (minutes < 60) return `${minutes}分钟前`;
  if (hours < 24) return `${hours}小时前`;
  if (days < 7) return `${days}天前`;
  
  return date.toLocaleString('zh-CN', {
    year: 'numeric',
    month: '2-digit',
    day: '2-digit',
  });
};

/**
 * 启动定时刷新
 */
const startAutoRefresh = () => {
  // 每分钟刷新未读数量
  refreshTimer = setInterval(() => {
    loadUnreadCount();
  }, 60000);
};

/**
 * 停止定时刷新
 */
const stopAutoRefresh = () => {
  if (refreshTimer) {
    clearInterval(refreshTimer);
    refreshTimer = null;
  }
};

// 组件挂载
onMounted(() => {
  loadNoticeList();
  loadUnreadCount();
  startAutoRefresh();
});

// 组件卸载
onUnmounted(() => {
  stopAutoRefresh();
});
</script>
```

### 通知列表组件

**文件：** `src/components/NoticeList.jsx`

```javascript
import React, { useState, useEffect } from 'react';
import { noticeService } from '../services/noticeService';

const NoticeList = () => {
  const [loading, setLoading] = useState(false);
  const [noticeList, setNoticeList] = useState([]);
  const [currentPage, setCurrentPage] = useState(1);
  const [pageSize] = useState(10);
  const [total, setTotal] = useState(0);
  const [unreadCount, setUnreadCount] = useState(0);

  // TODO: 从认证上下文获取
  const enterpriseId = 'ENTERPRISE_001';
  const username = 'test_user';

  useEffect(() => {
    loadNoticeList();
    loadUnreadCount();

    // 设置定时刷新
    const timer = setInterval(() => {
      loadUnreadCount();
    }, 60000);

    return () => clearInterval(timer);
  }, [currentPage]);

  const loadNoticeList = async () => {
    setLoading(true);
    try {
      const data = await noticeService.fetchList({
        enterprise_id: enterpriseId,
        page_no: currentPage,
        page_size: pageSize,
      });
      setNoticeList(data.list || []);
      setTotal(data.totalCount || 0);
    } catch (error) {
      console.error('加载通知列表失败:', error);
    } finally {
      setLoading(false);
    }
  };

  const loadUnreadCount = async () => {
    try {
      const count = await noticeService.fetchUnreadCount(username);
      setUnreadCount(count);
    } catch (error) {
      console.error('获取未读数量失败:', error);
    }
  };

  const handleViewNotice = (notice) => {
    // 导航到详情页
    window.location.href = `/notice/${notice.noticeId}`;
  };

  const formatDate = (dateStr) => {
    if (!dateStr) return '-';
    const date = new Date(dateStr);
    const now = new Date();
    const diff = now - date;
    const minutes = Math.floor(diff / 60000);
    const hours = Math.floor(diff / 3600000);
    const days = Math.floor(diff / 86400000);

    if (minutes < 1) return '刚刚';
    if (minutes < 60) return `${minutes}分钟前`;
    if (hours < 24) return `${hours}小时前`;
    if (days < 7) return `${days}天前`;

    return date.toLocaleDateString('zh-CN');
  };

  return (
    <div className="notice-list">
      <div className="page-header">
        <h1>通知中心</h1>
        {unreadCount > 0 && (
          <span className="badge">{unreadCount}</span>
        )}
      </div>

      {loading ? (
        <div className="loading">加载中...</div>
      ) : (
        <>
          {noticeList.length === 0 ? (
            <div className="empty">暂无通知</div>
          ) : (
            <div className="notice-items">
              {noticeList.map((notice) => (
                <div
                  key={notice.noticeId}
                  className={`notice-item ${notice.isImportant ? 'important' : ''}`}
                  onClick={() => handleViewNotice(notice)}
                >
                  {notice.isImportant === 1 && (
                    <span className="tag-important">重要</span>
                  )}
                  <h3>{notice.title}</h3>
                  <div className="meta">
                    <span>{notice.publisher}</span>
                    <span>{formatDate(notice.publishTime)}</span>
                  </div>
                </div>
              ))}
            </div>
          )}

          {total > pageSize && (
            <div className="pagination">
              <button
                disabled={currentPage === 1}
                onClick={() => setCurrentPage(currentPage - 1)}
              >
                上一页
              </button>
              <span>第 {currentPage} 页</span>
              <button
                disabled={currentPage * pageSize >= total}
                onClick={() => setCurrentPage(currentPage + 1)}
              >
                下一页
              </button>
            </div>
          )}
        </>
      )}
    </div>
  );
};

export default NoticeList;
```

## 移动端集成

### 使用 uni-app

```javascript
// utils/noticeApi.js
const API_BASE_URL = 'http://localhost:8082';

export const noticeApi = {
  /**
   * 获取通知列表
   */
  fetchList(params) {
    return new Promise((resolve, reject) => {
      uni.request({
        url: `${API_BASE_URL}/api/notice/list`,
        data: params,
        success: (res) => {
          if (res.data.code === 200) {
            resolve(res.data.data);
          } else {
            reject(new Error(res.data.message));
          }
        },
        fail: reject,
      });
    });
  },

  /**
   * 获取通知详情
   */
  fetchDetail(noticeId, params) {
    return new Promise((resolve, reject) => {
      uni.request({
        url: `${API_BASE_URL}/api/notice/${noticeId}`,
        data: params,
        success: (res) => {
          if (res.data.code === 200) {
            resolve(res.data.data);
          } else {
            reject(new Error(res.data.message));
          }
        },
        fail: reject,
      });
    });
  },
};
```

## 完整应用示例

### 包含所有功能的完整流程

```javascript
// store/noticeStore.js
import { defineStore } from 'pinia';
import { fetchNoticeList, fetchUnreadCount } from '@/api/notice';

export const useNoticeStore = defineStore('notice', {
  state: () => ({
    noticeList: [],
    unreadCount: 0,
    loading: false,
    currentPage: 1,
    pageSize: 10,
    total: 0,
  }),

  actions: {
    async loadNoticeList(enterpriseId) {
      this.loading = true;
      try {
        const data = await fetchNoticeList({
          enterprise_id: enterpriseId,
          page_no: this.currentPage,
          page_size: this.pageSize,
        });
        this.noticeList = data.list || [];
        this.total = data.totalCount || 0;
      } catch (error) {
        console.error('加载通知列表失败:', error);
        throw error;
      } finally {
        this.loading = false;
      }
    },

    async loadUnreadCount(username) {
      try {
        this.unreadCount = await fetchUnreadCount(username);
      } catch (error) {
        console.error('获取未读数量失败:', error);
      }
    },

    setCurrentPage(page) {
      this.currentPage = page;
    },
  },
});
```
